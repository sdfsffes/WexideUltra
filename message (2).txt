-- ========== ФИЛЬТР КОМАНДЫ ==========
local function IsTeammate(plr)
    if not TeamCheckEnabled then return false end
    if not plr or plr == Player then return true end
    if not plr.Team or not Player.Team then return false end
    if plr.Team == Player.Team then return true end
    return false
end

-- ========== ХИТБОКС ==========
local function CreateHitboxForCharacter(character)
    if not character then return end
    local hum = character:FindFirstChild("Humanoid")
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp or hum.Health <= 0 then return end
    
    if character:FindFirstChild("WexideHitbox") then
        character.WexideHitbox:Destroy()
    end
    
    local hitbox = Instance.new("Part")
    hitbox.Name = "WexideHitbox"
    hitbox.Size = Vector3.new(HitboxSize, HitboxSize, HitboxSize)
    hitbox.CFrame = hrp.CFrame
    hitbox.Anchored = true
    hitbox.CanCollide = true
    hitbox.CanQuery = true
    hitbox.CanTouch = true
    hitbox.Massless = true
    hitbox.Transparency = HitboxTransparency
    hitbox.Color = HitboxColor
    hitbox.Material = Enum.Material.Neon
    hitbox.Locked = true
    hitbox.Parent = character
    
    table.insert(HitboxObjects, hitbox)
    return hitbox
end

local function UpdateHitboxes()
    for _, obj in pairs(HitboxObjects) do
        if obj and obj.Parent then
            obj:Destroy()
        end
    end
    HitboxObjects = {}
    
    if not HitboxEnabled then return end
    
    for _, plr in pairs(game.Players:GetPlayers()) do
        if plr ~= Player and plr.Character then
            if IsTeammate(plr) then continue end
            
            local hum = plr.Character:FindFirstChild("Humanoid")
            if hum and hum.Health > 0 then
                CreateHitboxForCharacter(plr.Character)
            end
        end
    end
end

RunService.RenderStepped:Connect(function()

        if hitbox and hitbox.Parent then
            local char = hitbox.Parent

                hitbox.CFrame = hrp.CFrame

        end

end)

local function UpdateFOVCircle()
    if FOVCircleFrame then
        local size = AimbotFOV * 2
        FOVCircleFrame.Size = UDim2.new(0, size, 0, size)
        FOVCircleFrame.Position = UDim2.new(0.5, -AimbotFOV, 0.5, -AimbotFOV)
        
        for _, child in pairs(FOVCircleFrame:GetChildren()) do
            if child:IsA("Frame") then
                if child.BorderSizePixel > 0 then
                    child.BorderColor3 = FOVColor
                else
                    child.BackgroundColor3 = FOVColor
                end
            end
        end
    end
end

-- ========== ESP ==========
local function UpdateESP()
    for _, obj in pairs(ESPObjects) do
        if obj and obj.Parent then obj:Destroy() end
    end
    ESPObjects = {}
    if not ESPEnabled then return end
    
    for _, plr in pairs(game.Players:GetPlayers()) do
        if plr ~= Player and plr.Character then
            if IsTeammate(plr) then continue end
            
            local char = plr.Character
            local hum = char:FindFirstChild("Humanoid")
            if hum and hum.Health > 0 then
                local hl = Instance.new("Highlight")
                hl.Adornee = char
                hl.FillColor = ESPColor
                hl.FillTransparency = ESPTransparency
                hl.OutlineColor = ESPColor
                hl.OutlineTransparency = 0.2
                hl.Parent = char
                table.insert(ESPObjects, hl)
                
                local bill = Instance.new("BillboardGui")
                bill.Size = UDim2.new(0, 200, 0, 30)
                bill.Adornee = char:FindFirstChild("Head") or char:FindFirstChild("UpperTorso")
                bill.AlwaysOnTop = true
                bill.Parent = char
                
                local name = Instance.new("TextLabel")
                name.Size = UDim2.new(1, 0, 1, 0)
                name.BackgroundTransparency = 1
                name.Text = plr.Name
                name.TextColor3 = ESPColor
                name.TextSize = 16
                name.Font = Enum.Font.GothamBold
                name.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
                name.TextStrokeTransparency = 0.3
                name.Parent = bill
                table.insert(ESPObjects, bill)
            end
        end
    end
end

        if plr ~= Player and plr.Character then
            if IsTeammate(plr) then continue end
            
            local char = plr.Character
            local part = char:FindFirstChild("HumanoidRootPart")
            local hum = char:FindFirstChild("Humanoid")
            
            if part and hum and hum.Health > 0 then
                if HitboxEnabled and char:FindFirstChild("WexideHitbox") then
                    part = char.WexideHitbox
                end
                
                if VisibleCheck and not HitboxWallbang then
                    local ray = Ray.new(Camera.CFrame.Position, (part.Position - Camera.CFrame.Position).Unit * 1000)
                    local hit = game.Workspace:FindPartOnRayWithIgnoreList(ray, {Player.Character, Camera})
                    if hit and not hit:IsDescendantOf(char) and hit.Name ~= "WexideHitbox" then continue end
                end
                
                local distance = (part.Position - Camera.CFrame.Position).Magnitude
                if distance > AimbotDistance then continue end
                local screenPos, onScreen = Camera:WorldToScreenPoint(part.Position)
                if not onScreen then continue end
                local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
                local fovDist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                if fovDist < closestDistance then
                    closestDistance = fovDist
                    closest = plr
                end
            end
        end
    end
    return closest
end
